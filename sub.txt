finish on 6 cp:
	sectec
	coup_de_grace if fazed
	powder on 4+ targets
	eviscerate
build:
	backstab if premed
	shuriken_storm once if 7+ targets
	shadowstrike

# Outside Dance
finish on 6 cp:
	sectec during symbols
	rupture
	coup_de_grace if fazed
	powder on 3+ targets 
	eviscerate
build:
	backstab if flawless_form down
	shuriken_storm if 3+ targets
	backstab
